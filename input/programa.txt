int maximo(int a, int b) {
    int resultado = 0;
    if (a > b) {
        resultado = a;
    } else {
        resultado = b;
    }
    return resultado;
}

int x = 10;
int y = 20;
int limite = 100;
int acum = 0;
int i = 0;

while ((i < 5) && (acum < limite)) {
    acum = acum + x;
    i++;
}

int m = maximo(x, y);
int final = m + acum;
