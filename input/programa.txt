int suma(int a, int b) {
    int resultado = a + b;
    return resultado;
}

int factorial(int n) {
    int resultado = 1;
    int i = 1;
    while (i <= n) {
        resultado = resultado * i;
        i = i + 1;
    }
    return resultado;
}
